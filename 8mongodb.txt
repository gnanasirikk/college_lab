1. Find products priced above 500:

db.products.find({ price: { $gt: 500 } })


2. Find products with stock less than 10:

db.products.find({ stock: { $lt: 10 } })


3. Find products with price between 100 and 500:

db.products.find({ price: { $gte: 100, $lte: 500 } })



4. Find products not in category "electronics":

db.products.find({ category: { $ne: "electronics" } })


5. Find products in "electronics" or "furniture" categories:

db.products.find({ category: { $in: ["electronics", "furniture"] } })



6. Find products that are "electronics" and cost more than 1000:

db.products.find({ category: "electronics", price: { $gt: 1000 } })


7. Find products that are neither "food" nor "clothing":

db.products.find({ category: { $nin: ["food", "clothing"] } })


8. Find products with names starting with "S":

db.products.find({ name: { $regex: "^S" } })


9. Skip first 5 products and show next 10:

db.products.find().skip(5).limit(10)


10. Find products with a discount field:

db.products.find({ discount: { $exists: true } })
