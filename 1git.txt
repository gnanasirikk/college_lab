STEP 7: CONNECT LOCAL TO REMOTE

STEP 8: PUSH TO GITHUB

git branch -M main
git push -u origin main


STEP 9: EXPLORE FETCH

git fetch origin

# This downloads latest changes but does not merge


STEP 10: EXPLORE PULL

git pull origin main

# This fetches + merges changes
